# Эталонные ответы с разбором. Откройте после своей попытки.
# Проверить их можно так: LS_ANSWERS=solutions/answers.txt ./check.sh

# LV1 · LOOKER
01 ls
# Несколько каталогов — несколько аргументов; ls печатает каждый под заголовком «имя:».
02 ls etc logs
# Без -1 в терминале будут колонки. В конвейере (ls | ...) ls и так пишет по одному имени.
03 ls -1
# Шаблон раскрывает shell, а не ls: ls получает готовый список etc/host.conf etc/nsswitch.conf ...
# resolv.conf.bak не подходит: имя кончается на .bak.
04 ls etc/*.conf
# ? — ровно один любой символ: ?a?? — четыре символа, второй a.
05 ls words/?a??
# [ab] — один символ из набора.
06 ls words/[ab]*

# LV2 · DOT-FILE DETECTIVE
# -a показывает всё, что начинается с точки, включая . (этот каталог) и .. (родитель).
07 ls -a
# -A — то же без . и .. Удобно, когда считаете файлы.
08 ls -A
# -F: / каталог, * исполняемый, @ ссылка, | канал, = сокет.
09 ls -F
# -p ставит только /.
10 ls -p
# */ раскрывается в список каталогов (и ссылок на каталоги), -d не даёт ls заходить внутрь.
11 ls -d */

# LV3 · LONG-LIST LEGEND
12 ls -l
# -h без -l ничего не меняет: у ls без -l и -s нет колонки размеров.
13 ls -lh
# Номера inode одинаковые: notes.txt и hard.txt — два имени одного файла (счётчик ссылок 2).
14 ls -li notes.txt hard.txt
15 ls -R deep

# LV4 · SORT SORCERER
16 ls -lt logs
# -r переворачивает любую сортировку: свежие оказываются внизу, рядом с приглашением.
17 ls -ltr logs
18 ls -lS logs
19 ls -lSr logs
# Заглавная X. Строчная -x — совсем другое: имена строками, а не колонками.
20 ls -X docs
# -U — порядок, в котором записи лежат в каталоге. ls -f тоже не сортирует, но включает -a.
21 ls -U logs

# LV5 · THE LORD OF THE LiSt
# Ссылка, указанная аргументом, вместе с -l не разыменовывается: видна сама ссылка.
# ls -l current/ (со слэшем) показал бы уже содержимое releases/v2.
22 ls -l current
# В конвейере ls пишет по имени на строку, wc -l их считает. ls | wc -l пропустил бы скрытые.
23 ls -A | wc -l

# POWER COMBOS (входят в LV5)
24 ls -lah
# Без -d ls -l */ показал бы содержимое каждого каталога.
25 ls -ld */
# Первая строка ls -l — «total N», поэтому head -5 даёт только четыре файла.
26 ls -lhS logs | head -5
