# Эталонные ответы с разбором. Откройте после своей попытки.
# Проверить их можно так: LS_ANSWERS=solutions/answers.txt ./check.sh

# LV1 · LOOKER
01 ls
# Несколько каталогов — несколько аргументов; ls печатает каждый под заголовком «имя:».
02 ls etc logs
# Без -1 в терминале будут колонки. В конвейере (ls | ...) ls и так пишет по одному имени.
03 ls -1
# Шаблон раскрывает shell, а не ls: ls получает готовый список etc/host.conf etc/nsswitch.conf ...
# resolv.conf.bak не подходит: имя кончается на .bak.
04 ls etc/*.conf
# ? — ровно один любой символ: ?a?? — четыре символа, второй a.
05 ls words/?a??
# [bc] — один символ из набора.
06 ls words/[bc]*

# LV2 · DOT-FILE DETECTIVE
# -a показывает всё, что начинается с точки, включая . (этот каталог) и .. (родитель).
07 ls -a
# -A — то же без . и .. Удобно, когда считаете файлы.
08 ls -A
# -F: / каталог, * исполняемый, @ ссылка, | канал, = сокет.
09 ls -F
# -p ставит только /.
10 ls -p
# */ раскрывается в список каталогов (и ссылок на каталоги), -d не даёт ls заходить внутрь.
11 ls -d */
# -I прячет записи по шаблону. Кавычки нужны, чтобы *.gz раскрыл не shell, а ls.
12 ls -I '*.gz' logs

# LV3 · LONG-LIST LEGEND
13 ls -l
# -h без -l ничего не меняет: у ls без -l и -s нет колонки размеров.
14 ls -lh
# Без -d ls -l logs показал бы содержимое logs.
15 ls -ld logs
# Номера inode одинаковые: notes.txt и hard.txt — два имени одного файла (счётчик ссылок 2).
16 ls -li notes.txt hard.txt
# Числа нужны, когда владельца уже нет в /etc/passwd (после userdel) или он из другой системы.
17 ls -n notes.txt
# Колонка слева — место на диске в блоках по 1K: файл в 500 байт всё равно занимает 4K.
18 ls -s logs
# -L — разыменовать ссылку: размер и права цели, а не самой ссылки.
19 ls -lL link-to-notes
20 ls -l --time-style=long-iso logs
# --full-time = -l --time-style=full-iso.
21 ls --full-time logs

# LV4 · SORT SORCERER
22 ls -lt logs
# -r переворачивает любую сортировку: свежие оказываются внизу, рядом с приглашением.
23 ls -ltr logs
24 ls -lS logs
25 ls -lSr logs
26 ls -X docs
27 ls -v chapters
# -c с -lt: сортировка по ctime и ctime в колонке даты. ctime — время setup.sh, поэтому дата сегодняшняя.
28 ls -ltc logs
# -u с -lt: то же по atime — времени последнего чтения.
29 ls -ltu logs
# -U — порядок, в котором записи лежат в каталоге. ls -f тоже не сортирует, но включает -a.
30 ls -U logs
31 ls --group-directories-first
32 ls -R deep

# LV5 · LORD OF THE LiSts
# Ссылка, указанная аргументом, вместе с -l не разыменовывается: видна сама ссылка.
33 ls -l current
# Слэш в конце — путь уже не к ссылке, а к каталогу, на который она указывает.
34 ls -l current/
# В конвейере ls пишет по имени на строку, wc -l их считает. ls | wc -l пропустил бы скрытые.
35 ls -A | wc -l
# 4096 у каталога — размер самого каталога (списка имён), а не содержимого.
36 du -sh bigdir
# Кавычки ls ставит в терминале и сам ('my file.txt'), но это только отображение; -Q — всегда "...".
37 ls -Q
# Порядок сортировки задаёт локаль: в en_US apple Banana, в C — Banana apple (по кодам байтов).
38 LC_ALL=C ls
# alias ll — 'ls -alF' в ~/.bashrc Ubuntu. \ls или command ls запускают ls в обход алиаса.
39 ls -alF
# Первая строка ls -l — «total N», поэтому для трёх файлов нужно head -4.
40 ls -lS logs | head -4
